  <script>

    const videoUrl =
      document.getElementById("videoUrl");

    const loading =
      document.getElementById("loading");

    const result =
      document.getElementById("result");

    const error =
      document.getElementById("error");


    function processVideo() {

      const url =
        videoUrl.value.trim();

      result.style.display = "none";
      error.style.display = "none";


      if (!url) {

        showError(
          "Please paste a video URL."
        );

        return;
      }


      try {

        new URL(url);

      } catch {

        showError(
          "Please enter a valid video URL."
        );

        return;
      }


      loading.style.display = "block";


      /*
        DEMO PROCESSING

        Connect your permitted backend/API
        here for real video processing.
      */

      setTimeout(() => {

        loading.style.display = "none";

        document.getElementById(
          "videoTitle"
        ).textContent = "Video Ready";


        document.getElementById(
          "platformName"
        ).textContent =
          "Available download options";


        result.style.display = "block";

      }, 1800);

    }


    function showError(message) {

      error.textContent = message;

      error.style.display = "block";

    }


    function scrollToDownloader() {

      document
        .getElementById("downloader")
        .scrollIntoView({
          behavior: "smooth"
        });

    }


    function downloadVideo() {

      alert(
        "Demo mode: Connect Download.LK to a permitted backend/API to enable real downloads."
      );

    }


    /* ENTER KEY */

    videoUrl.addEventListener(
      "keydown",
      function(event) {

        if (event.key === "Enter") {

          processVideo();

        }

      }
    );


    /* QUALITY BUTTONS */

    document
      .querySelectorAll(".quality button")
      .forEach(button => {

        button.addEventListener(
          "click",
          function() {

            document
              .querySelectorAll(".quality button")
              .forEach(btn =>
                btn.classList.remove("active")
              );

            this.classList.add("active");

          }
        );

      });

  </script>
